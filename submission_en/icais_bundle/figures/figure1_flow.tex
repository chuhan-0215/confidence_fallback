  \textbf{Structure router}\\[-1pt]

\node[widemod, right=4mm of route] (coco) {%

\node[narrowmod, right=7mm of gate] (out) {%
  \textbf{Output}\\[-1pt]
  $y$%
};

\caption{\textbf{confidence\_fallback} interacts with the ProsQA inference environment. The input $x$ contains a directed graph and a reachability query (e.g., which candidate target is reachable from $r$ within $n_0$ latent steps). The agent routes $n_0$ from BFS depth, performs one Coconut forward pass to obtain $\hat y_0$ and $h_{n_0}$, and queries the M2 stop head for $p_0=\sigma(g(h_{n_0},n_0,x))$. If $p_0\ge\tau$, it returns $\hat y_0$; otherwise it invokes the kNN--M2 fallback path.}
