\begin{algorithm}[H]
\caption{Confidence Fallback}
\label{alg:conf-fallback}
\footnotesize
\begin{algorithmic}[1]
\Require $x$, $\tau$; Coconut $f$; stop head $g$
\Ensure Answer $\hat{y}$
\State $n_0 \gets \max\!\bigl(n_{\min}, \min(\mbox{\textsc{BfsDepth}}(x), n_{\max})\bigr)$
\State $(\hat{y}_0, h_{n_0}) \gets f(x, n_0)$
\State $s_0 \gets \mbox{\textsc{StabFeat}}(\hat{y}_0, n_0)$
\State $p_0 \gets \sigma\bigl(g(h_{n_0}, n_0, s_0)\bigr)$
\If{$p_0 \ge \tau$}
    \State \Return $\hat{y}_0$
\Else
    \State \Return \mbox{\Call{NeighborhoodRefine}{$x, f, g, \tau$}}
\EndIf
\end{algorithmic}
\end{algorithm}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\linewidth}
\begin{algorithm}[H]
\caption{Neighborhood Refinement}
\label{alg:refinement}
\footnotesize
\begin{algorithmic}[1]
\Require $x$; $f$; $g$; floor $n_{\mathrm{fl}}$; $\tau$
\Ensure Answer $\hat{y}$
\For{$n = n_{\mathrm{fl}}, \ldots, n_{\max}$}
    \State $(\hat{y}, h_n) \gets f(x, n)$
    \State $s_n \gets \mbox{\textsc{StabFeat}}(\hat{y}, n)$
    \State $p \gets \sigma\bigl(g(h_n, n, s_n)\bigr)$
    \If{$p \ge \tau$}
        \State \textbf{break}
    \EndIf
\EndFor
\State \Return $\hat{y}$
\end{algorithmic}
\end{algorithm}
\end{minipage}
