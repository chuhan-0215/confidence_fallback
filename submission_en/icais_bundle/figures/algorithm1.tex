\captionof{algorithm}{Confidence Fallback}
\label{alg:cf}
\scriptsize
\begin{algorithmic}[1]
\Require $x$, $\tau$, Coconut $f$, stop head $g$
\State $n_0 \leftarrow \max(n_{\min},\min(\textsc{BfsDepth}(x),n_{\max}))$;\ $(\hat y_0,h_{n_0}) \leftarrow f(x,n_0)$
\State $p_0 \leftarrow \sigma(g(h_{n_0},n_0,s_0))$ with $s_0=\textsc{StabFeat}(\hat y_0,n_0)$
\If{$p_0 \ge \tau$} \Return $\hat y_0$ \Else\ \Return first $\hat y_n$ from Eq.~\eqref{eq:policy} \EndIf
\end{algorithmic}
\end{minipage}
